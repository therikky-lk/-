cat > ~/install2.sh << 'INSTALL_END'
#!/bin/bash
# Установка WordCraft (Word), GridCraft (Excel) и DeckCraft (PowerPoint) от ArtCraft на Astra Linux 1.7
# Запуск:  bash install2.sh
# Программам нужна GLIBC 2.35+, а в Astra 1.7 стоит 2.28. Поэтому новая glibc
# распаковывается в ~/glibc (НЕ устанавливается в систему) и программы
# запускаются через неё. Сама система не меняется.

DL=~/Загрузки
APPS=~/craft
GL=~/glibc
mkdir -p "$DL" "$APPS" "$GL" ~/Desktops/Desktop1 ~/.local/share/applications
cd "$DL" || exit 1

# Скачать файл, если его ещё нет в «Загрузках»
get() {
  [ -s "$2" ] && return 0
  echo "Скачиваю $2 ..."
  if ! wget -q --show-progress -O "$2" "$1"; then
    rm -f "$2"
    echo "ОШИБКА: не удалось скачать $2"
    echo "Скачайте его в браузере по ссылке и положите в папку Загрузки:"
    echo "  $1"
    return 1
  fi
}

# 1. glibc 2.36 из Debian 12 (только распаковка)
if [ ! -f "$GL/lib/x86_64-linux-gnu/ld-linux-x86-64.so.2" ]; then
  DEB=$(ls libc6_2.36-*_amd64.deb 2>/dev/null | sort -V | tail -1)
  if [ -z "$DEB" ]; then
    DEB=libc6_2.36-9+deb12u14_amd64.deb
    get "https://mirror.yandex.ru/debian/pool/main/g/glibc/$DEB" "$DEB" || {
      echo "Если ссылка не работает, возьмите файл libc6_2.36-9+deb12u..._amd64.deb"
      echo "с самой большой цифрой тут: https://mirror.yandex.ru/debian/pool/main/g/glibc/"
      exit 1
    }
  fi
  dpkg-deb -x "$DEB" "$GL" || exit 1
fi
echo "glibc готова: $GL"

# 2. Общий запускатель
cat > ~/craft-run.sh << 'EOF'
#!/bin/bash
G=~/glibc/lib/x86_64-linux-gnu
exec "$G/ld-linux-x86-64.so.2" --library-path "$G:/lib/x86_64-linux-gnu:/usr/lib/x86_64-linux-gnu" "$@"
EOF
chmod +x ~/craft-run.sh

# 3. Установка одной программы: id, название, ссылка, файл в bin, категория
install_app() {
  local id=$1 title=$2 url=$3 bin=$4 cat=$5
  local file=${url##*/}
  local dir=$APPS/${file%.tar.gz}
  get "$url" "$file" || return 1
  tar -xzf "$file" -C "$APPS" || return 1
  local icon=$dir/share/icons/hicolor/256x256/apps/ai.storyteller.$id.png
  local desk=~/Desktops/Desktop1/$id.desktop
  cat > "$desk" << EOF
[Desktop Entry]
Type=Application
Name=$title
Exec=$HOME/craft-run.sh $dir/bin/$bin %f
Icon=$icon
Terminal=false
Categories=$cat
EOF
  chmod +x "$desk"
  cp "$desk" ~/.local/share/applications/
  echo "Готово: $title"
}

install_app wordcraft "WordCraft (Word)" \
  https://github.com/storytold/wordcraft/releases/download/v0.8.0/wordcraft-0.8.0-linux-x86_64.tar.gz \
  wordcraft "Office;WordProcessor;"

install_app gridcraft "GridCraft (Excel)" \
  https://github.com/storytold/gridcraft/releases/download/v0.8.0/gridcraft-0.8.0-linux-x86_64.tar.gz \
  gridcraft "Office;Spreadsheet;"

install_app deckcraft "DeckCraft (PowerPoint)" \
  https://github.com/storytold/deckcraft/releases/download/v0.8.0/deckcraft-0.8.0-linux-x86_64.tar.gz \
  deckcraft "Office;Presentation;"

# 4. После скачки: значки запускают программы через bash и пишут ошибки в ~/craft.log
#    (уже переделанные значки пропускаются, чтобы не сломать их повторно)
for f in ~/Desktops/Desktop1/{artcraft-launcher,photocraft,figcraft,wordcraft,gridcraft,deckcraft}.desktop; do
  [ -f "$f" ] || continue
  grep -q '^Exec=/bin/bash' "$f" && continue
  id=$(basename "$f" .desktop)
  cmd=$(grep '^Exec=' "$f" | sed 's/^Exec=//; s/ %f$//')
  printf '#!/bin/bash\n%s >> ~/craft.log 2>&1\n' "$cmd" > ~/run-$id.sh
  chmod +x ~/run-$id.sh
  sed -i "s|^Exec=.*|Exec=/bin/bash $HOME/run-$id.sh|" "$f"
  cp "$f" ~/.local/share/applications/
done

echo
echo "Всё. Значки на рабочем столе и в меню «Пуск»."
echo "Если значков не видно: правый клик по рабочему столу → «Обновить»."
echo "Если программа не открывается, ошибки смотрите командой: cat ~/craft.log"
INSTALL_END
bash ~/install2.sh
